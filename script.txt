    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance == part or result.Instance:IsDescendantOf(part.Parent)
end
local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if not IsEnemy(player) then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = GetAimPart(player)
    if not part then return false end
    if (part.Position - Camera.CFrame.Position).Magnitude > 3000 then return false end
    if Config.VisibleCheck and not HasLineOfSight(part) then return false end
    return true
end
local function AcquireTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local best, bestDist = nil, Config.FOV
    for _, player in ipairs(Players:GetPlayers()) do
        if IsEnemy(player) and player.Character then
            local part = GetAimPart(player)
            if part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bestDist then
                        if Config.VisibleCheck then
                            if HasLineOfSight(part) then bestDist = d; best = player end
                        else
                            bestDist = d; best = player
                        end
                    end
                end
            end
        end
    end
    return best
end
local function AimAt(player, instant)
    local part = GetAimPart(player)
    if not part then return end
    local goal = CFrame.new(Camera.CFrame.Position, part.Position)
    if instant or Config.Smoothness >= 1 then
        Camera.CFrame = goal
    else
        Camera.CFrame = Camera.CFrame:Lerp(goal, Config.Smoothness)
    end
end
local function IsCameraOnPart(part, thresholdDeg)
    if not part then return false end
    local camLook = Camera.CFrame.LookVector
    local toTarget = (part.Position - Camera.CFrame.Position)
    if toTarget.Magnitude < 0.01 then return true end
    toTarget = toTarget.Unit
    local dot = math.clamp(camLook:Dot(toTarget), -1, 1)
    return math.deg(math.acos(dot)) <= (thresholdDeg or 3)
end
local function FireOnce()
    pcall(function()
        if mouse1click then mouse1click()
        elseif mouse1press and mouse1release then
            mouse1press(); task.wait(0.02); mouse1release()
        end
    end)
end

local function nameMatchesKiller(name)
    if not name then return false end
    local n = string.lower(tostring(name))
    for _, kw in ipairs(KILLER_WORDS) do if string.find(n, kw, 1, true) then return true end end
    return false

local KILLER_ATTRS = {"IsKiller","Killer","IsMonster","Monster","IsSlasher","Slasher","IsMurderer","Murderer","IsRole","Role","Team","RoleName","PlayerRole"}
local function checkAttributes(obj)
    if not obj then return false end
    for _, attr in ipairs(KILLER_ATTRS) do
        local ok, v = pcall(function() return obj:GetAttribute(attr) end)
        if ok and v ~= nil then
            if v == true then return true end
            if type(v) == "string" and nameMatchesKiller(v) then return true end
            if type(v) == "number" and v == 1 then return true end
        end
    end
    return false
end
local KILLER_TAGS = {"Killer","killer","Monster","monster","Slasher","slasher","Vampire","Murderer"}
local function checkTags(obj)
    if not obj then return false end
    for _, tag in ipairs(KILLER_TAGS) do
        local ok, has = pcall(function() return CollectionService:HasTag(obj, tag) end)
        if ok and has then return true end
    end
    return false
end
local function isKillerPlayer(player)
    if not player then return false end
    if checkAttributes(player) then return true end
    if checkTags(player) then return true end
    if nameMatchesKiller(player.Name) then return true end
    if nameMatchesKiller(player.DisplayName) then return true end
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if nameMatchesKiller(v.Name) then return true end
            if v:IsA("StringValue") and nameMatchesKiller(v.Value) then return true end
        end
    end
    local char = player.Character
    if char then
        if nameMatchesKiller(char.Name) then return true end
        if checkAttributes(char) then return true end
        if checkTags(char) then return true end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if checkAttributes(hum) then return true end
            if checkTags(hum) then return true end
        end
        for _, obj in ipairs(char:GetDescendants()) do
            if nameMatchesKiller(obj.Name) then return true end
            if checkAttributes(obj) then return true end
            if checkTags(obj) then return true end
        end
    end
    return false
end

local function findRandomByKeywords(keywords)
    local found, seen = {}, {}
    local function addPart(p)
        if p and p:IsA("BasePart") and not seen[p] then seen[p]=true; table.insert(found,p) end
    end
    local function matchName(n)
        n = string.lower(n or "")
        for _, kw in ipairs(keywords) do if string.find(n, kw, 1, true) then return true end end
        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and matchName(obj.Name) then addPart(obj) end
        if obj:IsA("Model") and matchName(obj.Name) then
            if obj.PrimaryPart then addPart(obj.PrimaryPart)
            else for _,c in ipairs(obj:GetDescendants()) do if c:IsA("BasePart") then addPart(c); break end end end
        end
    end
    if #found == 0 then return nil end
    return found[math.random(1, #found)]
end
local function findKillerPartRandom()
    local killerPlayers = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and isKillerPlayer(p) then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then table.insert(killerPlayers, hrp) end
        end
    end
    if #killerPlayers > 0 then return killerPlayers[math.random(1, #killerPlayers)] end
    local killerParts = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        local isKiller = false
        if obj:IsA("Model") then
            if nameMatchesKiller(obj.Name) or checkAttributes(obj) or checkTags(obj) then isKiller = true
            else for _, c in ipairs(obj:GetChildren()) do if nameMatchesKiller(c.Name) or checkAttributes(c) or checkTags(c) then isKiller = true; break end end end
        end
        if isKiller then
            local base = obj.PrimaryPart or obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildWhichIsA("BasePart")
            if base then table.insert(killerParts, base) end
        elseif obj:IsA("BasePart") and (nameMatchesKiller(obj.Name) or checkAttributes(obj) or checkTags(obj)) then
            table.insert(killerParts, obj)
        end
    end
    if #killerParts > 0 then return killerParts[math.random(1, #killerParts)] end
    return nil
end

    flyBV.Name="MolFlyBV"; flyBV.MaxForce=Vector3.new(math.huge,math.huge,math.huge)
    flyBV.P=1250; flyBV.Velocity=Vector3.zero; flyBV.Parent=hrp
    flyBG = Instance.new("BodyGyro")
    flyBG.Name="MolFlyBG"; flyBG.MaxTorque=Vector3.new(math.huge,math.huge,math.huge)
    flyBG.P=100000; flyBG.D=1000; flyBG.CFrame=hrp.CFrame; flyBG.Parent=hrp
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.RenderStepped:Connect(SafeCall(function(dt)
        if not states.Fly then return end
        if not hrp.Parent or not flyBV or not flyBV.Parent or not flyBG or not flyBG.Parent then return end
        hum.PlatformStand=true
        if hum.WalkSpeed ~= 0 then hum.WalkSpeed=0 end
        if hum.JumpPower ~= 0 then hum.JumpPower=0 end
        if hum.AutoRotate then hum.AutoRotate=false end
        local camCF = Camera.CFrame
        local camLookFull = camCF.LookVector
        local camLookFlat = Vector3.new(camLookFull.X, 0, camLookFull.Z)
        if camLookFlat.Magnitude < 0.001 then camLookFlat = Vector3.new(0,0,-1) end
        camLookFlat = camLookFlat.Unit
        local camRightFlat = Vector3.new(camCF.RightVector.X, 0, camCF.RightVector.Z)
        if camRightFlat.Magnitude < 0.001 then camRightFlat = camLookFlat:Cross(Vector3.new(0,1,0)) end
        camRightFlat = camRightFlat.Unit
        local move = Vector3.zero
        local hasInput = false
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += camLookFull; hasInput=true end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= camLookFull; hasInput=true end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= camRightFlat; hasInput=true end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += camRightFlat; hasInput=true end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0,1,0); hasInput=true end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0,1,0); hasInput=true end
        local md = hum.MoveDirection
        if md.Magnitude > 0.05 then
            local camSpace = camCF:VectorToObjectSpace(md.Unit)
            local forwardAmt = -camSpace.Z; local rightAmt = camSpace.X
            local analogMove = Vector3.zero
            if math.abs(forwardAmt) > 0.01 then analogMove += camLookFull * forwardAmt end
            if math.abs(rightAmt) > 0.01 then analogMove += camRightFlat * rightAmt end
            if analogMove.Magnitude > 0.01 then move += analogMove.Unit * md.Magnitude; hasInput=true end
        end
        if hasInput and move.Magnitude > 0.01 then
            local vel = move.Unit * Config.FlySpeed
            flyBV.Velocity = vel
            pcall(function() hrp.AssemblyLinearVelocity = vel end)
        else
            flyBV.Velocity = Vector3.zero
            pcall(function() hrp.AssemblyLinearVelocity = Vector3.zero end)
            pcall(function() hrp.Velocity = Vector3.zero end)
        end
        if Config.Superman then
            if hasInput and move.Magnitude > 0.05 then
                local targetCF = CFrame.new(hrp.Position, hrp.Position + move.Unit)
                flyBG.CFrame = flyBG.CFrame:Lerp(targetCF, 0.4)
            else
                flyBG.CFrame = flyBG.CFrame:Lerp(camCF, 0.2)
            end
        else
            flyBG.CFrame = camCF
        end
    end, "FlyLoop"))
end

    local char = LocalPlayer.Character

        hum.PlatformStand = false
        if flySavedWS and flySavedWS > 0 then hum.WalkSpeed = flySavedWS end
        if flySavedJP and flySavedJP > 0 then hum.JumpPower = flySavedJP end
        hum.AutoRotate = (flySavedAutoRotate == nil) and true or flySavedAutoRotate
    end
end

local speedHookInstalled = false
local function installSpeedHook()
    if speedHookInstalled then return end
    speedHookInstalled = true
    if getrawmetatable and setreadonly and newcclosure then
        pcall(function()
            local mt = getrawmetatable(game)
            local oldIndex = mt.__index
            local oldNewIndex = mt.__newindex
            setreadonly(mt, false)
            mt.__index = newcclosure(function(self, key)
                if Config.Speed50 and key == "WalkSpeed" and self:IsA("Humanoid") and self.Parent == LocalPlayer.Character then
                    return Config.SpeedValue
                end
                return oldIndex(self, key)
            end)
            mt.__newindex = newcclosure(function(self, key, value)
                if Config.Speed50 and key == "WalkSpeed" and self:IsA("Humanoid") and self.Parent == LocalPlayer.Character then
                    return oldNewIndex(self, key, Config.SpeedValue)
                end
                return oldNewIndex(self, key, value)
            end)
            setreadonly(mt, true)
        end)
    end
end
installSpeedHook()

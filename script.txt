local currentTarget = nil

local function IsEnemy(player)
    if player == LocalPlayer then return false end
    if not Config.TeamCheck then return true end
    if player.Team and LocalPlayer.Team then return player.Team ~= LocalPlayer.Team end
    return true
end

local function GetAimPart(player)
    if not player or not player.Character then return nil end
    local char = player.Character
    local part = char:FindFirstChild(Config.AimPart)
    if part and part:IsA("BasePart") then return part end
    for _, name in ipairs({"Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso"}) do
        local p = char:FindFirstChild(name)
        if p and p:IsA("BasePart") then return p end
    end
    return nil
end

local function HasLineOfSight(part)
    local origin = Camera.CFrame.Position
    local dir = (part.Position - origin)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {LocalPlayer.Character, Camera}
    if part.Parent then table.insert(ignore, part.Parent) end
    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance == part or result.Instance:IsDescendantOf(part.Parent)
end

local function IsTargetValid(player)
    if not player or not player.Parent then return false end
    if not IsEnemy(player) then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = GetAimPart(player)
    if not part then return false end
    if (part.Position - Camera.CFrame.Position).Magnitude > 3000 then return false end
    if Config.VisibleCheck and not HasLineOfSight(part) then return false end
    return true
end

local function AcquireTarget()
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local best, bestDist = nil, Config.FOV
    for _, player in ipairs(Players:GetPlayers()) do
        if IsEnemy(player) and player.Character then
            local part = GetAimPart(player)
            if part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bestDist then
                        if Config.VisibleCheck then
                            if HasLineOfSight(part) then bestDist = d; best = player end
                        else
                            bestDist = d; best = player
                        end
                    end
                end
            end
        end
    end
    return best
end

local function AimAt(player, instant)
    local part = GetAimPart(player)
    if not part then return end
    local goal = CFrame.new(Camera.CFrame.Position, part.Position)
    if instant or Config.Smoothness >= 1 then
        Camera.CFrame = goal
    else
        Camera.CFrame = Camera.CFrame:Lerp(goal, Config.Smoothness)
    end
end

local function IsCameraOnPart(part, thresholdDeg)
    if not part then return false end
    local camLook = Camera.CFrame.LookVector
    local toTarget = (part.Position - Camera.CFrame.Position)
    if toTarget.Magnitude < 0.01 then return true end
    toTarget = toTarget.Unit
    local dot = math.clamp(camLook:Dot(toTarget), -1, 1)
    local angle = math.deg(math.acos(dot))
    return angle <= (thresholdDeg or 3)
end

    pcall(function()
        if mouse1click then
            mouse1click()
        elseif mouse1press and mouse1release then
            mouse1press(); task.wait(0.02); mouse1release()
        end
    end)
end

local function ClearLock() currentTarget = nil end

RunService.RenderStepped:Connect(SafeCall(function()
    if Config.AimbotEnabled then
        FOVCircle.Visible = true
        FOVCircle.Position = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        FOVCircle.Radius = Config.FOV
    else
        FOVCircle.Visible = false
    end

    local mouseHeld = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    local manualActive = Config.AimbotEnabled and mouseHeld
    local autoActive   = Config.AimbotEnabled and Config.AutoAim

    if not manualActive and not autoActive then
        if currentTarget then ClearLock() end
        return
    end

    if not Config.LockOn then
        currentTarget = AcquireTarget()
    end

    if currentTarget and not IsTargetValid(currentTarget) then ClearLock() end

    if not currentTarget then
        local nt = AcquireTarget()
        if nt then
            currentTarget = nt
            lastTargetSwitch = tick()
        end
    end

    if not currentTarget then return end

    local part = GetAimPart(currentTarget)
    if not part then ClearLock(); return end

    AimAt(currentTarget, false)

    if Config.AutoFire then
        if IsCameraOnPart(part, 3) then
            local now = tick()
            if now - lastFire >= 0.08 then
                lastFire = now
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, part.Position)
                FireOnce()
            end
        end
    end
end, "Aim"))

local KILLER_WORDS = {"killer","monster","zombie","slasher","hunter","stalker",
                     "chaser","entity","beast","demon","ghost","phantom","murderer"}

local function nameMatchesKiller(name)
    if not name then return false end
    local n = string.lower(tostring(name))
    for _, kw in ipairs(KILLER_WORDS) do
        if string.find(n, kw, 1, true) then return true end
    end
    return false
end

local function isKillerPlayer(player)
    if not player then return false end
    local ok, v = pcall(function() return player:GetAttribute("IsKiller") end)
    if ok and v == true then return true end
    ok, v = pcall(function() return player:GetAttribute("Killer") end)
    if ok and v == true then return true end
    ok, v = pcall(function() return player:GetAttribute("IsMonster") end)
    if ok and v == true then return true end
    if nameMatchesKiller(player.Name) then return true end
    if nameMatchesKiller(player.DisplayName) then return true end
    local char = player.Character
    if char then
        if nameMatchesKiller(char.Name) then return true end
        ok, v = pcall(function() return char:GetAttribute("IsKiller") end)
        if ok and v == true then return true end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and nameMatchesKiller(hum.Name) then return true end
    end
    return false
end

local function isGeneratorObj(obj)
    if not obj then return false end
    return string.find(string.lower(obj.Name), "generator", 1, true) ~= nil
end

    local found, seen = {}, {}
    local function addPart(p)
        if p and p:IsA("BasePart") and not seen[p] then seen[p]=true; table.insert(found,p) end
    end
    local function matchName(n)
        n = string.lower(n or "")
        for _, kw in ipairs(keywords) do
            if string.find(n, kw, 1, true) then return true end
        end
        return false
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and matchName(obj.Name) then addPart(obj) end
        if obj:IsA("Model") and matchName(obj.Name) then
            if obj.PrimaryPart then addPart(obj.PrimaryPart)
            else for _,c in ipairs(obj:GetDescendants()) do if c:IsA("BasePart") then addPart(c); break end end end
        end
        if obj:IsA("Folder") and matchName(obj.Name) then
            for _,c in ipairs(obj:GetDescendants()) do if c:IsA("BasePart") then addPart(c); break end end
        end
    end
    if #found == 0 then return nil end
    return found[math.random(1, #found)]
end

local function findKillerPart()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and isKillerPlayer(p) then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then return hrp end
        end
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and nameMatchesKiller(obj.Name) then
            local base = obj.PrimaryPart or obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildWhichIsA("BasePart")
            if base then return base end
        end
        if obj:IsA("BasePart") and nameMatchesKiller(obj.Name) then
            return obj
        end
        if obj:IsA("Model") then
            local ok, v = pcall(function() return obj:GetAttribute("IsKiller") end)
            if ok and v == true then
                local base = obj.PrimaryPart or obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildWhichIsA("BasePart")
                if base then return base end
            end
        end
    end
    return nil
end

local flyConn, flyBV, flyBG
local flySavedWS, flySavedJP, flySavedAutoRotate
local flyConflicts = {}

local function startFly()
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    flySavedWS = hum.WalkSpeed
    flySavedJP = hum.JumpPower
    flySavedAutoRotate = hum.AutoRotate

    hum.PlatformStand = true
    hum.AutoRotate = false
    hum.WalkSpeed = 0
    hum.JumpPower = 0

    for _, m in ipairs(hrp:GetChildren()) do
        if m.Name == "MolFlyBV" or m.Name == "MolFlyBG" then m:Destroy() end
    end

    flyConflicts = {}
    for _, m in ipairs(hrp:GetChildren()) do
        if m:IsA("BodyVelocity") or m:IsA("BodyGyro") then
            table.insert(flyConflicts, {obj=m, enabled=m.Enabled})
            m.Enabled = false
        end
    end

    flyBV = Instance.new("BodyVelocity")
    flyBV.Name = "MolFlyBV"
    flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp

    flyBG = Instance.new("BodyGyro")
    flyBG.Name = "MolFlyBG"
    flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    flyBG.P = 100000
    flyBG.D = 1000
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp

    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.RenderStepped:Connect(function(dt)
        pcall(function()
            if not states.Fly then return end
            if not hrp.Parent then return end
            if not flyBV or not flyBV.Parent then return end
            if not flyBG or not flyBG.Parent then return end

            hum.PlatformStand = true
            if hum.WalkSpeed ~= 0 then hum.WalkSpeed = 0 end
            if hum.JumpPower ~= 0 then hum.JumpPower = 0 end
            if hum.AutoRotate then hum.AutoRotate = false end

            local camCF = Camera.CFrame
            local camLook = camCF.LookVector
            local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
            if flatLook.Magnitude < 0.05 then
                local hl = hrp.CFrame.LookVector
                flatLook = Vector3.new(hl.X, 0, hl.Z)
            end
            if flatLook.Magnitude > 0.0001 then flatLook = flatLook.Unit end

            local flatRight = Vector3.new(camCF.RightVector.X, 0, camCF.RightVector.Z)
            if flatRight.Magnitude < 0.001 then
                flatRight = flatLook:Cross(Vector3.new(0, 1, 0))
            end
            if flatRight.Magnitude > 0.0001 then flatRight = flatRight.Unit end

            local move = Vector3.zero
            local hasInput = false

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += flatLook; hasInput = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= flatLook; hasInput = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= flatRight; hasInput = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += flatRight; hasInput = true end

            local humMove = hum.MoveDirection
            if humMove.Magnitude > 0.05 then
                local fm = Vector3.new(humMove.X, 0, humMove.Z)
                if fm.Magnitude > 0.05 then
                    move += fm.Unit * 1.2
                    hasInput = true
                end
            end

            local pitch = camCF.LookVector.Y
            if math.abs(pitch) > 0.05 then
                move += Vector3.new(0, pitch, 0)
                hasInput = true
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                move += Vector3.new(0, 1, 0); hasInput = true
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift)
              or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                move -= Vector3.new(0, 1, 0); hasInput = true
            end

            local targetVel
            if hasInput and move.Magnitude > 0.01 then
                targetVel = move.Unit * Config.FlySpeed
            else
                targetVel = Vector3.zero
            end
            flyBV.Velocity = targetVel
            pcall(function() hrp.AssemblyLinearVelocity = targetVel end)
            pcall(function() hrp.Velocity = targetVel end)

            if Config.Superman then
                if hasInput and move.Magnitude > 0.05 then
                    local targetCF = CFrame.new(hrp.Position, hrp.Position + move.Unit)
                    flyBG.CFrame = flyBG.CFrame:Lerp(targetCF, 0.4)
                else
                    flyBG.CFrame = flyBG.CFrame:Lerp(camCF, 0.2)
                end
            else
                flyBG.CFrame = camCF
            end
        end)
    end)
end

local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if flyBV then flyBV:Destroy(); flyBV = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end
    for _, e in ipairs(flyConflicts) do
        pcall(function() if e.obj and e.obj.Parent then e.obj.Enabled = e.enabled end end)
    end
    flyConflicts = {}
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.PlatformStand = false
        if flySavedWS and flySavedWS > 0 then hum.WalkSpeed = flySavedWS end
        if flySavedJP and flySavedJP > 0 then hum.JumpPower = flySavedJP end
        hum.AutoRotate = (flySavedAutoRotate == nil) and true or flySavedAutoRotate
    end
end

local speedHookInstalled = false
local function installSpeedHook()
    if speedHookInstalled then return end
    speedHookInstalled = true
    pcall(function()
        local mt = getrawmetatable(game)
        local oldIndex = mt.__index
        local oldNewIndex = mt.__newindex
        setreadonly(mt, false)
        mt.__index = newcclosure(function(self, key)
            if Config.Speed50 and key == "WalkSpeed" and self:IsA("Humanoid")
               and self.Parent == LocalPlayer.Character then
                return Config.SpeedValue
            end
            return oldIndex(self, key)
        end)
        mt.__newindex = newcclosure(function(self, key, value)
            if Config.Speed50 and key == "WalkSpeed" and self:IsA("Humanoid")
               and self.Parent == LocalPlayer.Character then
                return oldNewIndex(self, key, Config.SpeedValue)
            end
            return oldNewIndex(self, key, value)
        end)
        setreadonly(mt, true)
    end)
end
installSpeedHook()
